   567→// Tiny spring engine shared by every component.
   568→
   569→// Springs, sim-time timers (so slow-mo slows them too), one rAF loop, and token readers.
   570→
   571→window.Motion = (() => {
   572→
   573→  // theme: shared by every page through localStorage, live-updated by the gallery
   574→
   575→  const setTheme = t => { document.documentElement.dataset.theme = t === 'light' ? 'light' : 'dark'; };
   576→
   577→  try { setTheme(localStorage.getItem('ms-theme')); } catch { setTheme('dark'); }
   578→
   579→  addEventListener('storage', e => { if (e.key === 'ms-theme') setTheme(e.newValue); });
   580→
   581→
   582→
   583→  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
   584→
   585→  const S = (x, k = 400, c = 30) => ({ x, v: 0, t: x, k, c });
   586→
   587→  const tune = (s, k, c) => { s.k = k; s.c = reduce ? 2 * Math.sqrt(k) : c; };
   588→
   589→  const step = (s, dt) => { const a = -s.k * (s.x - s.t) - s.c * s.v; s.v += a * dt; s.x += s.v * dt; };
   590→
   591→  const clamp = (v, a, b) => Math.max(a, Math.min(b, v));
   592→
   593→
   594→
   595→  // tokens from shared/tokens.css
   596→
   597→  const token = (name, fallback = '') =>
   598→
   599→    getComputedStyle(document.documentElement).getPropertyValue(name).trim() || fallback;
   600→
   601→  const num = (name, fallback) => parseFloat(token(name, String(fallback)));
   602→
   603→  const spring = (name, fallback) => token(name, fallback).split(/\s+/).map(Number);
   604→
   605→
   606→
   607→  let timeScale = 1;
   608→
   609→  const timers = [];
   610→
   611→  const after = (sec, fn) => { const t = { t: sec, fn }; timers.push(t); return t; };
   612→
   613→  const cancel = t => { const i = timers.indexOf(t); if (i > -1) timers.splice(i, 1); };
   614→
   615→
   616→
   617→  // slow motion, synced with a parent page when embedded
   618→
   619→  const slowListeners = [];
   620→
   621→  function setSlow(v, fromParent = false) {
   622→
   623→    timeScale = v ? .25 : 1;
   624→
   625→    slowListeners.forEach(f => f(v));
   626→
   627→    if (!fromParent && parent !== window) parent.postMessage({ type: 'motion:slow-changed', on: v }, '*');
   628→
   629→  }
   630→
   631→  addEventListener('message', e => {
   632→
   633→    if (e.data && e.data.type === 'motion:slow') setSlow(!!e.data.on, true);
   634→
   635→    if (e.data && e.data.type === 'motion:theme') setTheme(e.data.theme);
   636→
   637→  });
   638→
   639→
   640→
   641→  const comps = [];
   642→
   643→  let last = performance.now();
   644→
   645→  function loop(now) {
   646→
   647→    const dt = Math.min((now - last) / 1000, 1 / 30) * timeScale; last = now;
   648→
   649→    for (let i = timers.length - 1; i >= 0; i--) {
   650→
   651→      if ((timers[i].t -= dt) <= 0) timers.splice(i, 1)[0].fn();
   652→
   653→    }
   654→
   655→    const sub = 4;
   656→
   657→    for (let k = 0; k < sub; k++) comps.forEach(c => c.frame(dt / sub));
   658→
   659→    comps.forEach(c => c.render());
   660→
   661→    requestAnimationFrame(loop);
   662→
   663→  }
   664→
   665→  requestAnimationFrame(loop);
   666→
   667→
   668→
   669→  return {
   670→
   671→    S, tune, step, clamp, token, num, spring, after, cancel,
   672→
   673→    register: c => comps.push(c),
   674→
   675→    get timeScale() { return timeScale; },
   676→
   677→    get slow() { return timeScale !== 1; },
   678→
   679→    setSlow, onSlow: f => slowListeners.push(f),
   680→
   681→    svg: (p, sw = 1.8) => `<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="${sw}" stroke-linecap="round" stroke-linejoin="round">${p}</svg>`,
   682→
   683→  };
   684→
   685→})();

  1389→// Tiny spring engine shared by every component.
  1390→
  1391→// Springs, sim-time timers (so slow-mo slows them too), one rAF loop, and token readers.
  1392→
  1393→window.Motion = (() => {
  1394→
  1395→  // theme: shared by every page through localStorage, live-updated by the gallery
  1396→
  1397→  const setTheme = t => { document.documentElement.dataset.theme = t === 'light' ? 'light' : 'dark'; };
  1398→
  1399→  try { setTheme(localStorage.getItem('ms-theme')); } catch { setTheme('dark'); }
  1400→
  1401→  addEventListener('storage', e => { if (e.key === 'ms-theme') setTheme(e.newValue); });
  1402→
  1403→
  1404→
  1405→  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  1406→
  1407→  const S = (x, k = 400, c = 30) => ({ x, v: 0, t: x, k, c });
  1408→
  1409→  const tune = (s, k, c) => { s.k = k; s.c = reduce ? 2 * Math.sqrt(k) : c; };
  1410→
  1411→  const step = (s, dt) => { const a = -s.k * (s.x - s.t) - s.c * s.v; s.v += a * dt; s.x += s.v * dt; };
  1412→
  1413→  const clamp = (v, a, b) => Math.max(a, Math.min(b, v));
  1414→
  1415→
  1416→
  1417→  // tokens from shared/tokens.css
  1418→
  1419→  const token = (name, fallback = '') =>
  1420→
  1421→    getComputedStyle(document.documentElement).getPropertyValue(name).trim() || fallback;
  1422→
  1423→  const num = (name, fallback) => parseFloat(token(name, String(fallback)));
  1424→
  1425→  const spring = (name, fallback) => token(name, fallback).split(/\s+/).map(Number);
  1426→
  1427→
  1428→
  1429→  let timeScale = 1;
  1430→
  1431→  const timers = [];
  1432→
  1433→  const after = (sec, fn) => { const t = { t: sec, fn }; timers.push(t); return t; };
  1434→
  1435→  const cancel = t => { const i = timers.indexOf(t); if (i > -1) timers.splice(i, 1); };
  1436→
  1437→
  1438→
  1439→  // slow motion, synced with a parent page when embedded
  1440→
  1441→  const slowListeners = [];
  1442→
  1443→  function setSlow(v, fromParent = false) {
  1444→
  1445→    timeScale = v ? .25 : 1;
  1446→
  1447→    slowListeners.forEach(f => f(v));
  1448→
  1449→    if (!fromParent && parent !== window) parent.postMessage({ type: 'motion:slow-changed', on: v }, '*');
  1450→
  1451→  }
  1452→
  1453→  addEventListener('message', e => {
  1454→
  1455→    if (e.data && e.data.type === 'motion:slow') setSlow(!!e.data.on, true);
  1456→
  1457→    if (e.data && e.data.type === 'motion:theme') setTheme(e.data.theme);
  1458→
  1459→  });
  1460→
  1461→
  1462→
  1463→  const comps = [];
  1464→
  1465→  let last = performance.now();
  1466→
  1467→  function loop(now) {
  1468→
  1469→    const dt = Math.min((now - last) / 1000, 1 / 30) * timeScale; last = now;
  1470→
  1471→    for (let i = timers.length - 1; i >= 0; i--) {
  1472→
  1473→      if ((timers[i].t -= dt) <= 0) timers.splice(i, 1)[0].fn();
  1474→
  1475→    }
  1476→
  1477→    const sub = 4;
  1478→
  1479→    for (let k = 0; k < sub; k++) comps.forEach(c => c.frame(dt / sub));
  1480→
  1481→    comps.forEach(c => c.render());
  1482→
  1483→    requestAnimationFrame(loop);
  1484→
  1485→  }
  1486→
  1487→  requestAnimationFrame(loop);
  1488→
  1489→
  1490→
  1491→  return {
  1492→
  1493→    S, tune, step, clamp, token, num, spring, after, cancel,
  1494→
  1495→    register: c => comps.push(c),
  1496→
  1497→    get timeScale() { return timeScale; },
  1498→
  1499→    get slow() { return timeScale !== 1; },
  1500→
  1501→    setSlow, onSlow: f => slowListeners.push(f),
  1502→
  1503→    svg: (p, sw = 1.8) => `<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="${sw}" stroke-linecap="round" stroke-linejoin="round">${p}</svg>`,
  1504→
  1505→  };
